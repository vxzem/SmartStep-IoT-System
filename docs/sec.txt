                    <span class="user-row-mat">${u.mat_id || '—'}</span>
                    <div class="user-row-actions">
                        <button class="row-btn row-btn-del" onclick="deleteUser('${u.id}', '${fullName}')">🗑 Remove</button>
                    </div>
                </div>`;
        }).join('');
    }

    // ── REGISTER USER ──
    async function registerUser() {
        const fname = document.getElementById('reg-fname').value.trim();
        const lname = document.getElementById('reg-lname').value.trim();
        const email = document.getElementById('reg-email').value.trim();
        const matId = document.getElementById('reg-matid').value.trim();

        const feedback = document.getElementById('regFeedback');
        const btn = document.getElementById('registerBtn');

        // Clear previous errors
        ['reg-fname','reg-lname','reg-email','reg-matid'].forEach(id => {
            document.getElementById(id).classList.remove('input-error');
        });
        feedback.style.display = 'none';

        // Validate fields
        let errors = [];
        if (!fname) { document.getElementById('reg-fname').classList.add('input-error'); errors.push('First name'); }
        if (!lname) { document.getElementById('reg-lname').classList.add('input-error'); errors.push('Last name'); }
        if (!email || !email.includes('@')) { document.getElementById('reg-email').classList.add('input-error'); errors.push('Valid email'); }
        if (!matId) { document.getElementById('reg-matid').classList.add('input-error'); errors.push('Mat ID'); }

        if (errors.length) {
            showFeedback('error', `Required: ${errors.join(', ')}.`);
            return;
        }

        btn.disabled = true;
        btn.textContent = 'Verifying...';

        try {
            // ── STEP 0: Verify mat_id + user_email exist in mat_management_settings ──
            const { data: matRecord, error: matErr } = await _supabase
                .from('mat_management_settings')
                .select('mat_id, user_email, user_id')
                .eq('mat_id', matId)
                .single();

            if (matErr || !matRecord) {
                showModal(
                    'Mat ID Not Found',
                    `The Mat ID you entered does not exist in the system. Only pre-registered mats can be assigned to users.`,
                    `Mat ID: ${matId}`
                );
                return;
            }

            // Check email matches what's on record for this mat
            if (matRecord.user_email?.toLowerCase() !== email.toLowerCase()) {
                showModal(
                    'Email Mismatch',
                    `The email address does not match the one on record for this mat. Please use the email that was registered with this Mat ID.`,
                    `Mat ID: ${matId}\nExpected email: ${matRecord.user_email || '(none on record)'}\nEntered email: ${email}`
                );
                return;
            }

            // Check mat is not already assigned to a user
            if (matRecord.user_id) {
                showModal(
                    'Mat Already Assigned',
                    `This mat is already linked to an existing user account. Each mat can only be assigned to one user.`,
                    `Mat ID: ${matId}`
                );
                return;
            }

            btn.textContent = 'Registering...';

            // ── STEP 1: Create auth user ──
            const { data: authData, error: authErr } = await _supabase.auth.signUp({
                email,
                password: crypto.randomUUID(),
                options: {
                    data: { first_name: fname, last_name: lname }
                }
            });

            if (authErr) throw authErr;

            const newUid = authData?.user?.id;
            if (!newUid) throw new Error('User creation failed — no UID returned.');

            // ── STEP 2: Upsert profile row ──
            const { error: profileErr } = await _supabase.from('profiles').upsert({
                id: newUid,
                first_name: fname,
                last_name: lname,
                mat_id: matId,
                role: 'user',
            });

            if (profileErr) throw profileErr;

            // ── STEP 3: Link user_id back into mat_management_settings ──
            const { error: mmsErr } = await _supabase
                .from('mat_management_settings')
                .update({
                    user_id: newUid,
                    current_status: 'active',
                    updated_at: new Date().toISOString(),
                })
                .eq('mat_id', matId);

            if (mmsErr) throw mmsErr;

            // Success
            showFeedback('success', `${fname} ${lname} registered successfully. A confirmation email has been sent to ${email}.`);
            showToast(`User ${fname} ${lname} added.`, 'success');

            ['reg-fname','reg-lname','reg-email','reg-matid'].forEach(id => {
                document.getElementById(id).value = '';
            });

            await loadUserList();

        } catch (err) {
            console.error(err);
            const msg = err.message?.includes('already registered')
                ? `Email "${email}" is already in use.`
                : err.message || 'Registration failed. Please try again.';
            showFeedback('error', msg);
            showToast('Registration failed.', 'error');
        } finally {
            btn.disabled = false;
            btn.textContent = '+ Add User to SmartStep';
        }
    }

    // ── DELETE USER ──
    async function deleteUser(uid, name) {
        if (!confirm(`Remove user "${name}" from SmartStep?\n\nThis will delete their profile. Their auth account may still exist.`)) return;

        const { error } = await _supabase.from('profiles').delete().eq('id', uid);
        if (error) { showToast('Failed to remove user: ' + error.message, 'error'); return; }

        const row = document.getElementById(`urow-${uid}`);
        if (row) {
            row.style.opacity = '0';
            row.style.transition = 'opacity 0.25s';
            setTimeout(() => { row.remove(); loadUserList(); }, 260);
        }
        showToast(`${name} removed.`, 'warn');
    }

        showToast('Cloud sync initiated. Data is up to date.', 'success');
    }

    // ── CSV EXPORT ──
    async function handleExportCSV() {
        const { data, error } = await _supabase
            .from('telemetry')
            .select('user_id, steps, total_wh, created_at')
            .order('created_at', { ascending: false });

        if (error || !data?.length) { showToast('No telemetry data to export.', 'warn'); return; }

        const headers = 'user_id,steps,total_wh,created_at';
        const rows = data.map(r => `${r.user_id},${r.steps},${r.total_wh},${r.created_at}`).join('\n');
        const csv = headers + '\n' + rows;

        const blob = new Blob([csv], { type: 'text/csv' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `smartstep-telemetry-${new Date().toISOString().slice(0,10)}.csv`;
        a.click();
        URL.revokeObjectURL(url);
        showToast('CSV downloaded.', 'success');
    }

    // ── SAVE PREFERENCES ──
    function saveAllPreferences() {
        const prefs = {
            units: document.getElementById('unitSelect').value,
            darkMode: document.getElementById('darkModeToggle').checked,
            refresh: document.getElementById('refreshSelect').value,
        };
        localStorage.setItem('adminPrefs', JSON.stringify(prefs));
        showToast('Preferences saved.', 'success');
    }

    function loadSavedPreferences() {
        try {
            const prefs = JSON.parse(localStorage.getItem('adminPrefs') || '{}');
            if (prefs.units) document.getElementById('unitSelect').value = prefs.units;
            if (prefs.darkMode !== undefined) document.getElementById('darkModeToggle').checked = prefs.darkMode;
            if (prefs.refresh) document.getElementById('refreshSelect').value = prefs.refresh;
        } catch {}
    }

    // ── FEEDBACK / TOAST ──
    function showFeedback(type, msg) {
        const el = document.getElementById('regFeedback');
        el.textContent = msg;
        el.className = `form-feedback ${type === 'success' ? 'feedback-success' : 'feedback-error'}`;
        el.style.display = 'block';
    }

    function showToast(msg, type = 'success') {
        const t = document.getElementById('toast');
        t.textContent = msg;
        t.className = `show t-${type}`;
        setTimeout(() => { t.className = ''; }, 3500);
    }
